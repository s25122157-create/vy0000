const SHEET_NAME = 'TelepathyData'; // 處理前端抓取遊戲狀態 (GET) function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (!sheet) { return createJsonResponse({ error: "找不到工作表: " + SHEET_NAME }); } var room = (e && e.parameter && e.parameter.room) ? e.parameter.room : "default"; var data = sheet.getDataRange().getValues(); var roomData = { status: "setup", question: "", answers: {} }; for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(room)) { roomData.status = data[i][1] || "setup"; roomData.question = data[i][2] || ""; try { roomData.answers = data[i][3] ? JSON.parse(data[i][3]) : {}; } catch (err) { roomData.answers = {}; } break; } } return createJsonResponse(roomData); } // 處理前端更新遊戲狀態或送出答案 (POST) function doPost(e) { // 取得腳本鎖定，防止多位玩家同時送出答案時互相覆蓋 var lock = LockService.getScriptLock(); try { lock.waitLock(10000); // 最多等待 10 秒 } catch (err) { return createJsonResponse({ success: false, error: "伺服器忙碌中，請稍後再試" }); } try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (!sheet) { return createJsonResponse({ success: false, error: "找不到工作表: " + SHEET_NAME }); } var req = JSON.parse(e.postData.contents); var room = req.room || "default"; var data = sheet.getDataRange().getValues(); var rowIndex = -1; var currentAnswers = {}; // 尋找房間並直接從 data 取得現有答案 for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(room)) { rowIndex = i + 1; try { currentAnswers = data[i][3] ? JSON.parse(data[i][3]) : {}; } catch (err) { currentAnswers = {}; } break; } } // 如果房間不存在，建立新房間 if (rowIndex === -1) { sheet.appendRow([room, "setup", "", "{}"]); rowIndex = sheet.getLastRow(); } // 根據 action 更新整列資料（減少多次 setValue 的效能損耗） if (req.action === "reset") { sheet.getRange(rowIndex, 2, 1, 3).setValues([["setup", "", "{}"]]); } else if (req.action === "set_question") { sheet.getRange(rowIndex, 2, 1, 3).setValues([["playing", req.question || "", "{}"]]); } else if (req.action === "submit_answer") { if (req.playerName) { currentAnswers[req.playerName] = req.answer; sheet.getRange(rowIndex, 4).setValue(JSON.stringify(currentAnswers)); } } else if (req.action === "reveal") { sheet.getRange(rowIndex, 2).setValue("revealed"); } return createJsonResponse({ success: true }); } catch (error) { return createJsonResponse({ success: false, error: error.toString() }); } finally { lock.releaseLock(); // 確保執行完畢一定釋放鎖定 } } // 統一回傳 JSON 格式的輔助函式 function createJsonResponse(payload) { return ContentService.createTextOutput(JSON.stringify(payload)) .setMimeType(ContentService.MimeType.JSON); }
